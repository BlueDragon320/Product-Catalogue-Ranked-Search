[KERNEL_MATH] Composite Scoring & Normalization Logic 
# 1. Directional Min-Max Normalization over Category Attribute Values:
if higher_is_better:
    norm_val = (val - min_val) / (max_val - min_val) if max_val != min_val else 1.0
else: # lower_is_better (e.g. Price, Latency, Weight)
    norm_val = (max_val - val) / (max_val - min_val) if max_val != min_val else 1.0

# 2. Multi-Criteria Composite Score (Strict invariant: sum(w_i) == 100.0%):
composite_score = sum( (weight_i / 100.0) * norm_val_i for i in active_attributes )

# 3. 3-Tier Deterministic Tie-Breaking Order:
ORDER BY composite_score DESC, created_at DESC, product_id ASC

# 4. Anti-Bias Sponsored Interleaving (Slots 1, 6, 11; max 20% cap):
interleaved_results = interleave_sponsored(organic_ranked, active_sponsored, interval=5)
